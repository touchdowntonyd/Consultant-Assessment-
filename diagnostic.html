<script>

  function showTab(tabId) {

    document.querySelectorAll('.tab-content').forEach(el => el.classList.remove('active'));

    document.querySelectorAll('.tab-btn').forEach(el => el.classList.remove('active'));

    document.getElementById(tabId).classList.add('active');

    event.currentTarget.classList.add('active');

  }


  function getSum(selector) {

    let sum = 0;

    document.querySelectorAll(selector).forEach(sel => {

      sum += parseInt(sel.value, 10) || 0;

    });

    return sum;

  }


  function runDiagnostics() {

    // 1. Calculate Moat Scores

    const scoreA = getSum('.sc-a');

    const scoreB = getSum('.sc-b');

    const scoreC = getSum('.sc-c');


    document.getElementById('sub-a').innerText = `${scoreA} / 20`;

    document.getElementById('sub-b').innerText = `${scoreB} / 20`;

    document.getElementById('sub-c').innerText = `${scoreC} / 20`;


    let dominantMoat = "Pillar A: High-Velocity Throughput";

    let prescribedVehicle = "Asset Production Tiers + Turnkey Sprints";

    let prescriptionText = "Your strength is rapid throughput. <strong>Stop selling open-ended hours or flat blended retainers.</strong> Package deliverables into standard asset batches with fixed turnaround SLAs. Institute an explicit 48-hour rush fee (+50%) and enforce revision limits.";


    if (scoreB > scoreA && scoreB >= scoreC) {

      dominantMoat = "Pillar B: Technical Infrastructure";

      prescribedVehicle = "Two-Phase Architecture (Audit Sprint + Usage Retainer)";

      prescriptionText = "Your strength is deep technical integration. <strong>Stop rolling implementation into monthly maintenance.</strong> Bill upfront setup as a dedicated engineering sprint (\$3,500–\$7,500), and tie monthly retainers to data/spend volume thresholds.";

    } else if (scoreC > scoreA && scoreC > scoreB) {

      dominantMoat = "Pillar C: Strategic Advisory";

      prescribedVehicle = "Advisory Retainer + Governance Seats";

      prescriptionText = "Your strength is partner-level executive counsel. <strong>Stop subsidizing client meetings out of production rates.</strong> Establish fixed monthly advisory retainers with capped steering sessions; unbundle tactical asset production into separate line items.";

    }


    // 2. Misalignment Bleed Score

    const leakScore = getSum('.leak-input');

    let leakLabel = "Operationally Aligned (Healthy)";

    let leakTag = "tag-green";


    if (leakScore > 12) {

      leakLabel = `Severe Dysmorphia (${leakScore}/20)`;

      leakTag = "tag-red";

    } else if (leakScore > 4) {

      leakLabel = `Moderate Margin Bleed (${leakScore}/20)`;

      leakTag = "tag-amber";

    }


    // 3. Client Leakage Forensics

    let totalLeakage = 0;

    const rows = document.querySelectorAll('#client-tbody tr');


    rows.forEach(row => {

      const fee = parseFloat(row.querySelector('.fee').value) || 0;

      const capHrs = parseFloat(row.querySelector('.cap-hrs').value) || 0;

      const actHrs = parseFloat(row.querySelector('.act-hrs').value) || 0;

      const costHr = parseFloat(row.querySelector('.cost-hr').value) || 0;


      const nomRate = capHrs > 0 ? (fee / capHrs) : 0;

      const ehrRate = actHrs > 0 ? (fee / actHrs) : 0;

      const netMargin = fee - (actHrs * costHr);

      const deltaPct = nomRate > 0 ? ((ehrRate - nomRate) / nomRate) * 100 : 0;


      row.querySelector('.nom-rate').innerText = `$${nomRate.toFixed(2)}/hr`;

      row.querySelector('.ehr-rate').innerText = `$${ehrRate.toFixed(2)}/hr`;

      row.querySelector('.net-margin').innerText = `$${netMargin.toLocaleString('en-US', {minimumFractionDigits: 2})}`;


      const deltaCell = row.querySelector('.margin-delta');

      deltaCell.innerText = `${deltaPct >= 0 ? '+' : ''}${deltaPct.toFixed(1)}%`;

      deltaCell.className = `num margin-delta ${deltaPct < -15 ? 'tag-red' : (deltaPct < 0 ? 'tag-amber' : 'tag-green')}`;


      // Sum leakage on scope creep (where actual hours exceed stated cap)

      if (actHrs > capHrs) {

        totalLeakage += (actHrs - capHrs) * costHr;

      }

    });


    // 4. Update Readout

    document.getElementById('out-moat').innerText = dominantMoat;

    document.getElementById('out-misalign').innerHTML = `<span class="tag ${leakTag}">${leakLabel}</span>`;

    document.getElementById('out-vehicle').innerText = prescribedVehicle;

    document.getElementById('out-leakage').innerText = `-$${totalLeakage.toLocaleString('en-US', {minimumFractionDigits: 2})} / mo`;

    document.getElementById('out-prescription').innerHTML = prescriptionText;

  }
